find_package(Qt6 REQUIRED COMPONENTS Test)

function(extwatch_add_test name)
    qt_add_executable(${name} ${name}.cpp testutil.h)
    target_link_libraries(${name} PRIVATE extwatch_core Qt6::Test extwatch_warnings)
    target_compile_definitions(${name} PRIVATE EXTWATCH_FIXTURES_DIR="${CMAKE_SOURCE_DIR}/fixtures")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

extwatch_add_test(test_crxid)

extwatch_add_test(test_inventory)
